# Package set for the common node image, one per line.
# Kept as data rather than a RUN line so a diff is readable.
#
# `node_exporter` is deliberately absent: EPEL 9 does not package it, so the
# Containerfile installs the pinned upstream release instead.

# boot and init
kernel
kernel-modules
dracut
dracut-network
dracut-config-generic
systemd
systemd-udev

# network
NetworkManager
iproute
iputils
ethtool
bridge-utils
# `curl-minimal` already ships in the base image and provides the binary
# Warewulf's initramfs module needs. Installing the full `curl` package
# would only fight it.
jq
rsync
chrony
openssh-server
openssh-clients

# hardware inventory. dmidecode and pciutils are used by the discovery image;
# the rest make the node self-describing after it boots.
dmidecode
pciutils
lshw
usbutils
smartmontools

# storage

parted

# `deployed` ref with; without it the node cannot configure itself.

python3-requests

# RDMA userspace.
#
# The `rdma_rxe` kernel module is already in the Rocky kernel this image
# installs; what the image lacked was the userspace to drive it. `rdma-core`
# brings the provider libraries and the udev rules, `libibverbs-utils` gives
# `ibv_devinfo` so a test can prove a verbs device exists before benchmarking
# it, and `perftest` is the benchmark itself (`ib_write_bw`, `ib_send_bw`).
#
# This is Soft-RoCE: a software RDMA data path over an ordinary Ethernet NIC.
# It is not InfiniBand hardware and there is no subnet manager. See
# `cluster-platform-wip/docs/rdma-soft-roce.md`.
rdma-core
libibverbs-utils
perftest
